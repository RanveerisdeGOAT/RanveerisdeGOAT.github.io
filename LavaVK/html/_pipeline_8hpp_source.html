562 struct RenderingInfo {
563 VkRect2D renderArea{};
564
565 uint32_t layerCount = 1;
566 uint32_t viewMask = 0;
567
568 std::vector<RenderingAttachmentInfo> colorAttachments;
569
570 std::optional<RenderingAttachmentInfo> depthAttachment;
571 std::optional<RenderingAttachmentInfo> stencilAttachment;
572
573
574 std::vector<VkRenderingAttachmentInfo> vkColorAttachments;
575 VkRenderingAttachmentInfo vkDepthAttachment{};
576 VkRenderingAttachmentInfo vkStencilAttachment{};
577
578
579 VkRenderingInfo native() {
580 vkColorAttachments.clear();
581
582 for (const auto &attachment: colorAttachments) {
583 vkColorAttachments.push_back({
584 .sType = VK_STRUCTURE_TYPE_RENDERING_ATTACHMENT_INFO,
585 .imageView = attachment.image->view(),
586 .imageLayout = static_cast<VkImageLayout>(attachment.layout),
587 .loadOp = static_cast<VkAttachmentLoadOp>(attachment.loadOp),
588 .storeOp = static_cast<VkAttachmentStoreOp>(attachment.storeOp),
589 .clearValue = attachment.clearValue.value
590 });
591 }
592
593
594 if (depthAttachment) {
595 vkDepthAttachment = {
596 .sType = VK_STRUCTURE_TYPE_RENDERING_ATTACHMENT_INFO,
597 .imageView = depthAttachment->image->view(),
598 .imageLayout = static_cast<VkImageLayout>(depthAttachment->layout),
599 .loadOp = static_cast<VkAttachmentLoadOp>(depthAttachment->loadOp),
600 .storeOp = static_cast<VkAttachmentStoreOp>(depthAttachment->storeOp),
601 .clearValue = depthAttachment->clearValue.value
602 };
603 }
604
605
606 if (stencilAttachment) {
607 vkStencilAttachment = {
608 .sType = VK_STRUCTURE_TYPE_RENDERING_ATTACHMENT_INFO,
609 .imageView = stencilAttachment->image->view(),
610 .imageLayout = static_cast<VkImageLayout>(stencilAttachment->layout),
611 .loadOp = static_cast<VkAttachmentLoadOp>(stencilAttachment->loadOp),
612 .storeOp = static_cast<VkAttachmentStoreOp>(stencilAttachment->storeOp),
613 .clearValue = stencilAttachment->clearValue.value
614 };
615 }
616
617
618 return {
619 .sType = VK_STRUCTURE_TYPE_RENDERING_INFO,
620 .pNext = nullptr,
621 .flags = 0,
622 .renderArea = renderArea,
623 .layerCount = layerCount,
624 .viewMask = viewMask,
625
626 .colorAttachmentCount =
627 static_cast<uint32_t>(vkColorAttachments.size()),
628
629 .pColorAttachments =
630 vkColorAttachments.data(),
631
632 .pDepthAttachment =
633 depthAttachment ? &vkDepthAttachment : nullptr,
634
635 .pStencilAttachment =
636 stencilAttachment ? &vkStencilAttachment : nullptr
637 };
638 }
