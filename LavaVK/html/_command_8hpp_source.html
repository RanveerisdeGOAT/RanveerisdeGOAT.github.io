430 barrier.subresourceRange.baseArrayLayer = 0;
431 barrier.subresourceRange.layerCount = 1;
432 barrier.srcAccessMask = static_cast<VkAccessFlags>(srcAccess);
433 barrier.dstAccessMask = static_cast<VkAccessFlags>(dstAccess);
